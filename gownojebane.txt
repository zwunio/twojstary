local function normalizeKeyCode(name)
	name = clean(name)
	if typeof(name) ~= "string" or name == "" or name:lower() == "none" then
		return Enum.KeyCode.Unknown
	end

	local ok, key = pcall(function()
		return Enum.KeyCode[name]
	end)
	if ok then return key end

	ok, key = pcall(function()
		return Enum.KeyCode[name:upper()]
	end)
	if ok then return key end

	return Enum.KeyCode.Unknown
end

local function createKeyPicker(parent, allocate, name, data)
	name = clean(name)
	data = data or {}

	local layoutOrder = allocate()
	local key = normalizeKeyCode(data.Default)
	local mode = clean(data.Mode or "Hold")
	if mode == "" then mode = "Hold" end

	local btn = Instance.new("TextButton")
	btn.LayoutOrder = layoutOrder
	btn.Size = UDim2.new(1, 0, 0, 34)
	btn.BackgroundColor3 = Color3.fromRGB(29, 30, 38)
	btn.BorderSizePixel = 0
	btn.TextColor3 = Color3.new(1, 1, 1)
	btn.Font = Enum.Font.GothamMedium
	btn.TextSize = 13
	btn.TextXAlignment = Enum.TextXAlignment.Left
	btn.Parent = parent

	local corner = Instance.new("UICorner")
	corner.CornerRadius = UDim.new(0, 8)
	corner.Parent = btn

	local picker = {
		Name = name,
		Key = key,
		Mode = mode,
		State = false,
		Button = btn,
		Callback = data.Callback,
	}

	function picker:Refresh()
		local keyName = picker.Key == Enum.KeyCode.Unknown and "None" or picker.Key.Name
		btn.Text = "  " .. clean(data.Text or name) .. "  [" .. keyName .. "]"
	end

	function picker:GetState()
		if picker.Mode == "Always" then
			return true
		end
		return picker.State == true
	end

	function picker:SetValue(v)
		picker.Key = normalizeKeyCode(v)
		picker.State = false
		picker:Refresh()
	end

	function picker:OnBegan(k)
		if picker.Key == Enum.KeyCode.Unknown or k ~= picker.Key then
			return
		end

		if picker.Mode == "Toggle" then
			picker.State = not picker.State
			if picker.Callback then
				pcall(picker.Callback, picker.State)
			end
		elseif picker.Mode == "Hold" or picker.Mode == "Always" then
			picker.State = true
			if picker.Callback then
				pcall(picker.Callback, true)
			end
		end
	end

	function picker:OnEnded(k)
		if picker.Key == Enum.KeyCode.Unknown or k ~= picker.Key then
			return
		end

		if picker.Mode == "Hold" and picker.State then
			picker.State = false
			if picker.Callback then
				pcall(picker.Callback, false)
			end
		end
	end

	btn.MouseButton1Click:Connect(function()
		activePicker = picker
		btn.Text = "  Press any key..."
	end)

	picker:Refresh()
	Options[name] = picker
	table.insert(keyPickers, picker)

	return picker
end

		fillCorner.CornerRadius = UDim.new(1, 0)
		fillCorner.Parent = fill

		local obj
		local dragging = false

		local function formatValue(v)
			if rounding == 0 then
				return tostring(math.floor(v + 0.5))
			else
				return string.format("%." .. rounding .. "f", v)
			end
		end

		local function clampRound(v)
			v = math.clamp(v, min, max)
			if rounding == 0 then
				return math.floor(v + 0.5)
			end
			local mult = 10 ^ rounding
			return math.floor(v * mult + 0.5) / mult
		end

		local function setValue(v, fire)
			value = clampRound(v)
			fill.Size = UDim2.fromScale(max > min and ((value - min) / (max - min)) or 0, 1)
			valueLabel.Text = formatValue(value)
			if obj then
				obj.Value = value
			end
			if fire and data.Callback then
				pcall(data.Callback, value)
			end
		end

		local function updateFromX(x)
			local absPos = track.AbsolutePosition.X
			local absSize = math.max(track.AbsoluteSize.X, 1)
			local rel = math.clamp((x - absPos) / absSize, 0, 1)
			local raw = min + rel * (max - min)
			setValue(raw, true)
		end

		track.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging = true
				updateFromX(input.Position.X)
			end
		end)

		UserInputService.InputChanged:Connect(function(input)
			if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
				updateFromX(input.Position.X)
			end
		end)

		UserInputService.InputEnded:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton1 then
				dragging = false
			end
		end)

		obj = {
			Value = value,
			SetValue = function(self, v)
				setValue(v, true)
			end,
		}

		Options[name] = obj
		setValue(value, false)

		if data.Callback then
			task.defer(function()
				if not Library.Unloaded then
					pcall(data.Callback, value)
				end
			end)
		end

		return obj
	end

local function isValidAimTarget(target)
	if not target or not target.Parent then
		return false
	end

	local humanoid = target:FindFirstChildOfClass("Humanoid")
	local root = target:FindFirstChild("HumanoidRootPart")
	return humanoid and root and humanoid.Health > 0
end

local function getAimPart(target)
	local preferred = _G.PrivateProjectAimPart or "HumanoidRootPart"
	return target:FindFirstChild(preferred)
		or target:FindFirstChild("HumanoidRootPart")
		or target:FindFirstChild("Head")
end

local function hasLineOfSight(target)
	if not AimbotWallCheck then
		return true
	end

	local root = target:FindFirstChild("HumanoidRootPart")
	if not root or not camera then
		return false
	end

	local origin = camera.CFrame.Position
	local direction = root.Position - origin

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {
		LocalPlayer.Character,
		target
	}

	local hit = Workspace:Raycast(origin, direction, params)
	return hit == nil
end

local function findClosestTarget(maxFOV)
	camera = Workspace.CurrentCamera
	if not camera then
		return nil
	end

	local center = Vector2.new(
		camera.ViewportSize.X * 0.5,
		camera.ViewportSize.Y * 0.5
	)

	local closest = nil
	local closestDistance = maxFOV or math.huge

	for _, target in ipairs(getTestTargets()) do
		if isValidAimTarget(target) and hasLineOfSight(target) then
			local root = getAimPart(target)
			if not root then
				continue
			end

			local maxDistance = _G.PrivateProjectMaxTargetDistance or math.huge
			if LocalPlayer.Character then
				local myRoot = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
				if myRoot and (root.Position - myRoot.Position).Magnitude > maxDistance then
					continue
				end
			end

			local screenPos, onScreen = camera:WorldToViewportPoint(root.Position)
			if onScreen and screenPos.Z > 0 then
				local distance = (
					Vector2.new(screenPos.X, screenPos.Y) - center
				).Magnitude

				if distance <= closestDistance then
					closestDistance = distance
					closest = target
				end
			end
		end
	end

	return closest
end

local function getSilentAimTarget()
	if not SilentAimEnabled then
		SilentAimTarget = nil
		return nil
	end

	SilentAimTarget = findClosestTarget(SilentAimFOV)
	return SilentAimTarget
end

_G.PrivateProjectGetSilentAimTarget = getSilentAimTarget

local function aimAtTarget(target, deltaTime)
	camera = Workspace.CurrentCamera
	if not camera or not isValidAimTarget(target) then
		return
	end

	local root = getAimPart(target)
	if not root then
		return
	end

	local desired = CFrame.lookAt(camera.CFrame.Position, root.Position)
	local alpha = 1 - math.pow(
		1 - math.clamp(AimbotSmoothness, 0.01, 1),
		deltaTime * 60
	)

	camera.CFrame = camera.CFrame:Lerp(desired, alpha)
end

local RagebotKeyHeld = false

local function getRageTargetPart(model)
	if not model then return nil end
	return model:FindFirstChild(RagebotPart)
		or model:FindFirstChild("Head")
		or model:FindFirstChild("HumanoidRootPart")
end

local function isRageValid(model)
	if not model or not model.Parent then return false end
	if model == LocalPlayer.Character then return false end

	local hum = model:FindFirstChildOfClass("Humanoid")
	local part = getRageTargetPart(model)
	if not hum or hum.Health <= 0 or not part then return false end

	local root = model:FindFirstChild("HumanoidRootPart") or part
	local myChar = LocalPlayer.Character
	local myRoot = myChar and (myChar:FindFirstChild("HumanoidRootPart") or myChar:FindFirstChild("Head"))
	if not myRoot then return false end

	if (root.Position - myRoot.Position).Magnitude > RagebotMaxDistance then return false end

	return true
end

local function rageVisible(part)
	if not RagebotWallCheck or not part or not camera then return true end

	local origin = camera.CFrame.Position
	local direction = part.Position - origin

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {LocalPlayer.Character, camera}

	local hit = Workspace:Raycast(origin, direction, params)
	return not hit or hit.Instance:IsDescendantOf(part.Parent)
end

local function collectRageTargets()
	local targets = {}

	local folder = Workspace:FindFirstChild("TestTargets")
	if folder then
		for _, child in ipairs(folder:GetChildren()) do
			if isRageValid(child) then table.insert(targets, child) end
		end
	end

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer and player.Character and isRageValid(player.Character) then
			table.insert(targets, player.Character)
		end
	end

	return targets
end

local function chooseRageTarget()
	local mouse = UserInputService:GetMouseLocation()
	local best, bestScore = nil, math.huge

	for _, target in ipairs(collectRageTargets()) do
		local part = getRageTargetPart(target)
		if part and rageVisible(part) then
			local point, onScreen = camera:WorldToViewportPoint(part.Position)
			if onScreen and point.Z > 0 then
				local screenDist = (Vector2.new(point.X, point.Y) - mouse).Magnitude
				if (not RagebotFOVEnabled or screenDist <= RagebotFOV) then
					local root = target:FindFirstChild("HumanoidRootPart") or part
					local myRoot = LocalPlayer.Character and (LocalPlayer.Character:FindFirstChild("HumanoidRootPart") or LocalPlayer.Character:FindFirstChild("Head"))
					local worldDist = myRoot and (root.Position - myRoot.Position).Magnitude or math.huge
					local score = RagebotPriority == "FOV" and screenDist or worldDist
					if score < bestScore then best, bestScore = target, score end
				end
			end
		end
	end

	return best
end

local function rageSnap(target)
	local part = getRageTargetPart(target)
	if not part or not camera then return end
	camera.CFrame = CFrame.lookAt(camera.CFrame.Position, part.Position)
end

RunService:BindToRenderStep(
	"PrivateProjectAim",
	Enum.RenderPriority.Camera.Value + 1,
	function(deltaTime)
		camera = Workspace.CurrentCamera

		if RagebotEnabled and Options.RagebotKey and Options.RagebotKey:GetState() then
			if RagebotSwitching or not isRageValid(RagebotTarget) then
				RagebotTarget = chooseRageTarget()
			end

			if RagebotTarget then
				rageSnap(RagebotTarget)
			end
		else
			RagebotTarget = nil
		end

		local effectiveFOV = AimbotFOVEnabled and AimbotFOV or math.huge

		FOVCircle.Visible = AimbotEnabled and AimbotFOVEnabled
		FOVCircle.Size = UDim2.fromOffset(AimbotFOV * 2, AimbotFOV * 2)

		if AimbotEnabled and (Options.AimbotKey and Options.AimbotKey:GetState()) then
			if not AimbotSwitching and isValidAimTarget(CurrentAimTarget) then
				-- Keep the current target.
			else
				CurrentAimTarget = findClosestTarget(effectiveFOV)
			end

			if CurrentAimTarget then
				aimAtTarget(CurrentAimTarget, deltaTime)
			end
		elseif not AimbotEnabled then
			CurrentAimTarget = nil
		end

		if SilentAimEnabled and (Options.TargetSelectorKey and Options.TargetSelectorKey:GetState()) then
			SilentAimTarget = findClosestTarget(SilentAimFOV)
		elseif not SilentAimEnabled then
			SilentAimTarget = nil
		end

		local markerTarget = SilentAimTarget or CurrentAimTarget
		if markerTarget and (_G.PrivateProjectShowTargetMarker ~= false) then
			local markerPart = getAimPart(markerTarget)
			if markerPart then
				local point, visible = camera:WorldToViewportPoint(markerPart.Position)
				TargetMarker.Visible = visible and point.Z > 0
				TargetMarker.Position = UDim2.fromOffset(point.X, point.Y)
			else
				TargetMarker.Visible = false
			end
		else
			TargetMarker.Visible = false
		end

		AimStatus:SetText(
			"Aimbot target: " ..
			(CurrentAimTarget and CurrentAimTarget.Name or "none")
		)

		SilentStatus:SetText(
			"Silent target: " ..
			(SilentAimTarget and SilentAimTarget.Name or "none")
		)
	end
)

local function getOrCreateEffect(className, name)
	className = clean(className)
	name = clean(name)

	local effect = Lighting:FindFirstChild(name)
	if effect and effect:IsA(className) then
		return effect
	end

	if effect then effect:Destroy() end

	effect = Instance.new(className)
	effect.Name = name
	effect.Parent = Lighting
	return effect
end

local function applyColorPreset(name)
	name = clean(name)

	if name == "Normal" then
		colorCorrection.Brightness = 0
		colorCorrection.Contrast = 0
		colorCorrection.Saturation = 0
		colorCorrection.TintColor = Color3.new(1, 1, 1)
	elseif name == "Cinematic" then
		colorCorrection.Brightness = -0.02
		colorCorrection.Contrast = 0.22
		colorCorrection.Saturation = -0.08
		colorCorrection.TintColor = Color3.fromRGB(235, 240, 255)
	elseif name == "Cold" then
		colorCorrection.Brightness = 0
		colorCorrection.Contrast = 0.12
		colorCorrection.Saturation = 0.05
		colorCorrection.TintColor = Color3.fromRGB(190, 220, 255)
	elseif name == "Warm" then
		colorCorrection.Brightness = 0.02
		colorCorrection.Contrast = 0.08
		colorCorrection.Saturation = 0.18
		colorCorrection.TintColor = Color3.fromRGB(255, 225, 190)
	elseif name == "Neon" then
		colorCorrection.Brightness = 0.03
		colorCorrection.Contrast = 0.25
		colorCorrection.Saturation = 0.45
		colorCorrection.TintColor = Color3.fromRGB(220, 255, 240)
	end
end

local function flashLightning()
	if not lightningEnabled or not visualsEnabled then return end

	LightningFlash.BackgroundTransparency = 0.72
	task.delay(0.05, function()
		if LightningFlash then LightningFlash.BackgroundTransparency = 1 end
	end)

	task.delay(0.12, function()
		if LightningFlash and lightningEnabled and visualsEnabled then
			LightningFlash.BackgroundTransparency = 0.88
			task.delay(0.07, function()
				if LightningFlash then LightningFlash.BackgroundTransparency = 1 end
			end)
		end
	end)
end

task.spawn(function()
	while not Library.Unloaded do
		if lightningEnabled and visualsEnabled then
			task.wait(RNG:NextNumber(lightningDelayMin, lightningDelayMax))
			flashLightning()
		else
			task.wait(1)
		end
	end
end)

	if VignetteStroke then VignetteStroke.Transparency = 1 end
	if FilmGrain then FilmGrain.TextTransparency = 1 end
	if LightningFlash then LightningFlash.BackgroundTransparency = 1 end
end

local function resetVisuals()
	rainEnabled = false
	snowEnabled = false
	fogEnabled = false
	bloomEnabled = false
	nightEnabled = false
	colorBoostEnabled = false
	sunEnabled = false
	dofEnabled = false
	vignetteEnabled = false
	lightningEnabled = false
	blurEnabled = false
	colorCycleEnabled = false
	bloomPulseEnabled = false
	cameraSwayEnabled = false
	filmGrainEnabled = false

	setRain(false)

	SnowGui.Enabled = false
	bloom.Intensity = 0
	atmosphere.Density = 0
	atmosphere.Haze = 0
	sunRays.Intensity = 0
	depthOfField.FarIntensity = 0
	depthOfField.NearIntensity = 0
	blur.Size = 0

	colorCorrection.Brightness = 0
	colorCorrection.Contrast = 0
	colorCorrection.Saturation = 0
	colorCorrection.TintColor = Color3.new(1, 1, 1)

	Lighting.ExposureCompensation = originalLighting.ExposureCompensation
	Lighting.Brightness = originalLighting.Brightness
	Lighting.ClockTime = originalLighting.ClockTime
	Lighting.Ambient = originalLighting.Ambient
	Lighting.OutdoorAmbient = originalLighting.OutdoorAmbient
	Lighting.FogStart = originalLighting.FogStart
	Lighting.FogEnd = originalLighting.FogEnd
	Lighting.GlobalShadows = originalLighting.GlobalShadows

	camera.FieldOfView = originalCameraFOV

	VignetteStroke.Transparency = 1
	FilmGrain.TextTransparency = 1
	LightningFlash.BackgroundTransparency = 1

	local names = {
		"VisualsEnabled", "Rain", "Snow", "Fog", "NightMode", "Lightning",
		"ColorBoost", "Bloom", "SunRays", "DepthOfField", "Blur",
		"Vignette", "FilmGrain", "ColorCycle", "BloomPulse", "CameraSway"
	}

	for _, name in ipairs(names) do
		if Options[name] then
			pcall(function() Options[name]:SetValue(false) end)
		end
	end

	if Options.VisualsEnabled then
		pcall(function() Options.VisualsEnabled:SetValue(true) end)
	end

	if Options.ColorPreset then
		pcall(function() Options.ColorPreset:SetValue("Normal") end)
	end
end

RunService.RenderStepped:Connect(function(deltaTime)
	if SnowGui.Enabled then
		for _, flake in ipairs(snowflakes) do
			local p = flake.Position
			local y = p.Y.Scale + deltaTime * snowSpeed
			local x = p.X.Scale + math.sin(os.clock() * 0.8 + flake.AbsolutePosition.Y) * deltaTime * snowWind
			if y > 1.08 then
				y = -0.08
				x = RNG:NextNumber(0, 1)
			end
			flake.Position = UDim2.fromScale(x, y)
		end
	end

	if RainGui.Enabled then
		for _, drop in ipairs(rainDrops) do
			local pos = drop.Position
			local y = pos.Y.Scale + deltaTime * rainSpeed
			if y > 1.15 then
				drop.Position = UDim2.fromScale(RNG:NextNumber(-0.05, 1.05), RNG:NextNumber(-0.2, -0.05))
			else
				drop.Position = UDim2.fromScale(pos.X.Scale - deltaTime * rainWind, y)
			end
		end
	end

	if visualsEnabled then
		if colorCycleEnabled then
			local hue = (os.clock() * 0.035) % 1
			colorCorrection.TintColor = Color3.fromHSV(hue, 0.22, 1)
		end

		if bloomPulseEnabled and bloomEnabled then
			bloom.Intensity = 0.65 + math.sin(os.clock() * 2.2) * 0.35
		end

		if cameraSwayEnabled then
			local t = os.clock()
			local sway = math.sin(t * 1.15) * 0.0015
			camera.CFrame = camera.CFrame * CFrame.Angles(0, 0, sway)
		end

		if filmGrainEnabled then
			FilmGrain.Rotation = RNG:NextInteger(-2, 2)
			FilmGrain.TextTransparency = RNG:NextNumber(0.82, 0.93)
		end
	end
end)

local function makeESPPanel(target)
	local root = target:FindFirstChild("HumanoidRootPart")
	local humanoid = target:FindFirstChildOfClass("Humanoid")
	if not root or not humanoid then return end

	local guiPanel = Instance.new("BillboardGui")
	guiPanel.Name = "Advanced_" .. target.Name
	guiPanel.Adornee = root
	guiPanel.AlwaysOnTop = true
	guiPanel.Size = UDim2.fromOffset(240, 110)
	guiPanel.StudsOffset = Vector3.new(0, 3.4, 0)
	guiPanel.Parent = AdvancedESPFolder

	local frame = Instance.new("Frame")
	frame.BackgroundTransparency = 1
	frame.Size = UDim2.fromScale(1, 1)
	frame.Parent = guiPanel

	local box = Instance.new("Frame")
	box.Name = "Box"
	box.BackgroundTransparency = 1
	box.Size = UDim2.fromScale(0.62, 0.72)
	box.Position = UDim2.fromScale(0.19, 0.08)
	box.Visible = LabState.espShowBox
	box.Parent = frame

	local stroke = Instance.new("UIStroke")
	stroke.Thickness = LabState.espOutlineThickness
	stroke.Color = ESPColor
	stroke.Transparency = LabState.espOutlineTransparency
	stroke.Parent = box

	local healthBack = Instance.new("Frame")
	healthBack.Name = "HealthBack"
	healthBack.BorderSizePixel = 0
	healthBack.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
	healthBack.BackgroundTransparency = 0.15
	healthBack.Size = UDim2.fromOffset(5, 70)
	healthBack.Position = UDim2.fromScale(0.15, 0.08)
	healthBack.Visible = LabState.espShowHealthBar
	healthBack.Parent = frame

	local healthFill = Instance.new("Frame")
	healthFill.Name = "HealthFill"
	healthFill.BorderSizePixel = 0
	healthFill.BackgroundColor3 = Color3.fromRGB(80, 220, 120)
	healthFill.AnchorPoint = Vector2.new(0, 1)
	healthFill.Position = UDim2.fromScale(0, 1)
	healthFill.Size = UDim2.fromScale(1, 1)
	healthFill.Parent = healthBack

	local label = Instance.new("TextLabel")
	label.BackgroundTransparency = 1
	label.Size = UDim2.fromScale(1, 0.23)
	label.Position = UDim2.fromScale(0, 0.77)
	label.Font = Enum.Font.GothamBold
	label.TextSize = LabState.espTextSize
	label.TextColor3 = ESPTextColor
	label.TextStrokeTransparency = 0.25
	label.Text = target.Name
	label.Parent = frame

	local connection
	connection = RunService.RenderStepped:Connect(function()
		if not guiPanel.Parent or not target.Parent or humanoid.Health <= 0 then
			pcall(function() connection:Disconnect() end)
			if guiPanel.Parent then guiPanel:Destroy() end
			return
		end

		if not camera then return end

		local distance = (camera.CFrame.Position - root.Position).Magnitude
		if distance > LabState.espMaxDistance then
			guiPanel.Enabled = false
			return
		end

		guiPanel.Enabled = ESPEnabled

		local ratio = math.clamp(humanoid.Health / math.max(humanoid.MaxHealth, 1), 0, 1)
		healthFill.Size = UDim2.fromScale(1, ratio)
		healthFill.BackgroundColor3 = Color3.fromHSV(math.clamp(ratio * 0.33, 0, 0.33), 0.8, 1)

		box.Visible = LabState.espShowBox
		healthBack.Visible = LabState.espShowHealthBar
		stroke.Thickness = LabState.espOutlineThickness
		stroke.Color = LabState.espRainbow and Color3.fromHSV((os.clock() * 0.2) % 1, 0.8, 1) or ESPColor
		stroke.Transparency = LabState.espOutlineTransparency
		label.TextSize = LabState.espTextSize
		label.Text = target.Name .. "  |  " .. math.floor(distance + 0.5) .. "m  |  " .. math.floor(humanoid.Health + 0.5) .. " HP"
	end)

	table.insert(LabConnections, connection)
end

	for _, target in ipairs(getTestTargets()) do
		makeESPPanel(target)
	end
end

local CrosshairLabBox = StudioLabTab:AddRightGroupbox("Crosshair Lab")
local CrosshairParts = {}
local CrosshairColor = Color3.fromRGB(255, 255, 255)

local function clearCrosshairParts()
	for _, object in ipairs(CrosshairParts) do
		if object and object.Parent then object:Destroy() end
	end
	table.clear(CrosshairParts)
end

local function createCrosshairPart(name, position, size)
	local part = Instance.new("Frame")
	part.Name = name
	part.AnchorPoint = Vector2.new(0.5, 0.5)
	part.Position = position
	part.Size = size
	part.BorderSizePixel = 0
	part.BackgroundColor3 = CrosshairColor
	part.Parent = CrosshairGui
	table.insert(CrosshairParts, part)
	return part
end

local function rebuildCrosshair()
	clearCrosshairParts()

	if not LabState.crosshairEnabled then
		if Crosshair then Crosshair.Visible = false end
		return
	end

	if Crosshair then Crosshair.Visible = false end

	local center = UDim2.fromScale(0.5, 0.5)
	local s = LabState.crosshairSize
	local g = LabState.crosshairGap
	local t = LabState.crosshairThickness

	if LabState.crosshairStyle == "+" then
		createCrosshairPart("Top", UDim2.new(0.5, 0, 0.5, -(g + s / 2)), UDim2.fromOffset(t, s))
		createCrosshairPart("Bottom", UDim2.new(0.5, 0, 0.5, g + s / 2), UDim2.fromOffset(t, s))
		createCrosshairPart("Left", UDim2.new(0.5, -(g + s / 2), 0.5, 0), UDim2.fromOffset(s, t))
		createCrosshairPart("Right", UDim2.new(0.5, g + s / 2, 0.5, 0), UDim2.fromOffset(s, t))
	elseif LabState.crosshairStyle == "•" then
		createCrosshairPart("Dot", center, UDim2.fromOffset(math.max(2, t + 1), math.max(2, t + 1)))
	elseif LabState.crosshairStyle == "◆" then
		local dot = createCrosshairPart("Diamond", center, UDim2.fromOffset(math.max(4, t + 4), math.max(4, t + 4)))
		local corner = Instance.new("UICorner")
		corner.CornerRadius = UDim.new(0, 2)
		corner.Parent = dot
		dot.Rotation = 45
	elseif LabState.crosshairStyle == "┼" then
		createCrosshairPart("Horizontal", center, UDim2.fromOffset(s * 2 + g * 2, t))
		createCrosshairPart("Vertical", center, UDim2.fromOffset(t, s * 2 + g * 2))
	end
end

local function saveOriginalSky()
	if AethereaOriginalSky ~= nil then return end

	if current then
		AethereaOriginalSky = current:Clone()
	else
		AethereaOriginalSky = false
	end
end

local function restoreAethereaSky()
	local lab = Lighting:FindFirstChild("AethereaLabSky")
	if lab then lab:Destroy() end

	local current = Lighting:FindFirstChildOfClass("Sky")
	if current and current.Name == "AethereaLabSky" then current:Destroy() end

	if AethereaOriginalSky and typeof(AethereaOriginalSky) == "Instance" then
		local clone = AethereaOriginalSky:Clone()
		clone.Parent = Lighting
	end

	AethereaSky = nil
end

local counterYDistance = 100
local counterCooldown = 0.2

local function counterTeleport()
	local Character = LocalPlayer.Character
	if not Character then
		Character = LocalPlayer.CharacterAdded:Wait()
	end

	local Root = Character:WaitForChild("HumanoidRootPart")
	Root.CFrame = CFrame.new(Root.Position + Vector3.new(0, counterYDistance, 0))
end

	Text = "UE Counter",
	Tooltip = "Counters UE's ragebot",
	-- Executor-safe default: OFF so you don't fly on load.
	-- Flip to true if you want it armed at execution.
	Default = false,
	Callback = function(value)
		counterEnabled = value

		if counterEnabled then
			task.spawn(function()
				while counterEnabled and not Library.Unloaded do
					counterTeleport()
					task.wait(counterCooldown)
				end
			end)
		end
	end
})
